\section{Reference / Lookup Data}

Read endpoints are publicly accessible.
Write endpoints update existing entries in the reference tables.

\subsection{Appointed Subjects}
\apiendpoint{\GET}{/apointed-subjects}{\noauth}{Returns all appointed subjects.}
\apiendpoint{\PATCH}{/apointed-subjects/\{id\}}{\noauth}{Updates an appointed subject by ID.}

\subsection{Authorities}
\apiendpoint{\GET}{/authorities}{\noauth}{Returns all authorities.}
\apiendpoint{\PUT}{/authorities/\{id\}}{\noauth}{Updates an authority by ID.}

\subsection{Blood Groups}
\apiendpoint{\GET}{/blood-groups}{\noauth}{Returns all blood groups.}
\apiendpoint{\PUT}{/blood-groups/\{id\}}{\noauth}{Updates a blood group by ID.}

\subsection{Titles}
