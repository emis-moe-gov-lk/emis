\section{Versions \& Change Logs}

\apiendpoint{\GET}{/versions}{\noauth}{Returns all application versions together with their change logs.}
\apiendpoint{\POST}{/versions}{\noauth}{Creates a new version entry.}
\apiendpoint{\PUT}{/versions/\{id\}}{\noauth}{Updates a version entry.}
\apiendpoint{\DELETE}{/versions/\{id\}}{\noauth}{Deletes a version entry.}
