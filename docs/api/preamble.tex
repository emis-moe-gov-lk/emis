% -------------------------------------------------------
%  ENDPOINT MACRO
%
%  Each endpoint renders as its own tabularx row so it is
%  guaranteed to be a block-level element — no inline flow.
%
%  Visual output:
%
%    [METHOD]  /path/here                         [AUTH]
%              Description of what this does.
%    ─────────────────────────────────────────────────────
%
%  Usage: \apiendpoint{\GET}{/path}{\authjwt}{description}
% -------------------------------------------------------
\makeatletter
\gdef\ep@color{cpub}

\newcommand{\GET}{\gdef\ep@color{cget}\mbadge{cget}{GET}}
\newcommand{\POST}{\gdef\ep@color{cpost}\mbadge{cpost}{POST}}
\newcommand{\PUT}{\gdef\ep@color{cput}\mbadge{cput}{PUT}}
\newcommand{\PATCH}{\gdef\ep@color{cpatch}\mbadge{cpatch}{PATCH}}
\newcommand{\DELETE}{\gdef\ep@color{cdel}\mbadge{cdel}{DELETE}}

\newcommand{\apiendpoint}[4]{%
  \par\vspace{2pt}%
  \noindent
  \makebox[2.0cm][l]{#1}%
  \makebox[0.7\linewidth][l]{\texttt{\bfseries\small #2}}%
  \hfill#3\par
  \noindent\hspace{8pt}\parbox{0.95\linewidth}{\small\color{ctext}#4}\par
  \vspace{2pt}%
  \noindent\textcolor{cborder}{\rule{\linewidth}{0.5pt}}%
  \par\vspace{4pt}%
}
\makeatother
